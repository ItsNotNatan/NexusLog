    chave VARCHAR(100) PRIMARY KEY,
    valor TEXT NOT NULL

-- Tabela de Filiais
CREATE TABLE filiais (
    id VARCHAR(10) PRIMARY KEY, -- Ex: 'BR02', 'BR04', 'BR06'
    nome VARCHAR(100) NOT NULL,
    cidade VARCHAR(100) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabela de Usuários (Acesso Multi-Filial)
CREATE TABLE usuarios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    senha VARCHAR(50) NOT NULL,
    nome_completo VARCHAR(255) NOT NULL,
    cargo VARCHAR(50) DEFAULT 'OPERADOR' CHECK (cargo IN ('ADM', 'LIDER', 'OPERADOR')),
    filial_padrao_id VARCHAR(10) REFERENCES filiais(id) ON DELETE SET NULL,
    filiais_acesso TEXT[] DEFAULT '{}',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Catálogo de Materiais
CREATE TABLE materiais (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    desenho_sap VARCHAR(100),
    part_number VARCHAR(100),
    descricao VARCHAR(255) NOT NULL,
    fornecedor VARCHAR(150),
    unidade_medida VARCHAR(20) DEFAULT 'Unid',
    valor_unitario NUMERIC(15, 2) DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Estoque Físico
CREATE TABLE estoque (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    material_id UUID REFERENCES materiais(id) ON DELETE CASCADE,
    filial_id VARCHAR(10) REFERENCES filiais(id) ON DELETE CASCADE,
    
    desenho_sap VARCHAR(100), 
    part_number VARCHAR(100),
    descricao VARCHAR(255),

    fornecedor VARCHAR(150),
    referencia VARCHAR(100),
    unidade_medida VARCHAR(20) DEFAULT 'Unid',
    data_necessidade DATE,
    emissao_nf VARCHAR(20),
    receb_nf VARCHAR(20),
    valor_unitario NUMERIC(15, 2),
    centro VARCHAR(50),
    deposito VARCHAR(50),

    nf_entrada VARCHAR(100), 
    documento_compras VARCHAR(100),
    wbs VARCHAR(100),
    nome_projeto VARCHAR(255), -- ✨ ADICIONADO AQUI
    alocacao VARCHAR(100),
    
    is_transferencia BOOLEAN DEFAULT FALSE,
    
    quantidade_disponivel NUMERIC(15, 2) DEFAULT 0,
    quantidade_reservada NUMERIC(15, 2) DEFAULT 0, 
    status VARCHAR(50) DEFAULT 'Disponível' CHECK (status IN ('Disponível', 'Zerado')),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Histórico de Edições Manuais no Estoque
CREATE TABLE historico_edicoes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    estoque_id UUID REFERENCES estoque(id) ON DELETE CASCADE,
    usuario VARCHAR(255),
    campo_alterado VARCHAR(100),
    valor_antigo TEXT,
    valor_novo TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Solicitações
CREATE TABLE solicitacoes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ps VARCHAR(50) UNIQUE NOT NULL, -- Pedido de Solicitação (Gerado pelo sistema)
    pl VARCHAR(50) UNIQUE,          -- Packing List (Preenchido pelo Líder ao aprovar)
    
    usuario_id UUID REFERENCES usuarios(id) ON DELETE SET NULL,
    nome_solicitante VARCHAR(255), 
    tipo VARCHAR(50) NOT NULL CHECK (tipo IN ('Material', 'Transferencia WBS', 'Nota Fiscal', 'Entrada', 'Crossdocking', 'Reintegracao', 'Cancelado')),
    filial_origem_id VARCHAR(10) REFERENCES filiais(id) ON DELETE SET NULL,
    destino TEXT, 
    wbs_origem VARCHAR(100),
    wbs_destino VARCHAR(100),
    data_necessidade DATE,
    
    data_entrega TIMESTAMP WITH TIME ZONE, 
    
    observacoes TEXT,
    entrega_urgente BOOLEAN DEFAULT FALSE, 
    status VARCHAR(50) DEFAULT 'Pendente' CHECK (status IN ('Pendente', 'Em Separação', 'Concluído', 'Cancelado', 'Recusado')),
    
    data_aprovacao_pl TIMESTAMP WITH TIME ZONE,
    prazo_finalizacao_pl TIMESTAMP WITH TIME ZONE,
    
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Itens das Solicitações
CREATE TABLE solicitacoes_itens (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    solicitacao_id UUID REFERENCES solicitacoes(id) ON DELETE CASCADE, 
    material_id UUID REFERENCES materiais(id) ON DELETE SET NULL,
    estoque_id UUID REFERENCES estoque(id) ON DELETE SET NULL,

    desenho_sap_manual VARCHAR(100),
    part_number_manual VARCHAR(100),
    descricao_manual VARCHAR(255),
    quantidade_solicitada NUMERIC(15, 2) NOT NULL CHECK (quantidade_solicitada > 0),
    unidade_medida_manual VARCHAR(20),
    valor_unitario_manual NUMERIC(15, 2),
    
    fornecedor VARCHAR(150),
    referencia VARCHAR(100),
    nf_entrada VARCHAR(100),
    wbs_element VARCHAR(100),
    nome_projeto VARCHAR(255), -- ✨ ADICIONADO AQUI
    emissao_nf VARCHAR(20),
    receb_nf VARCHAR(20),
    documento_compras VARCHAR(100),
    centro VARCHAR(50),
    deposito VARCHAR(50),
    alocacao VARCHAR(100),
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Packing Lists
CREATE TABLE packing_lists (
    numero_pl SERIAL PRIMARY KEY, 
    solicitacao_id UUID REFERENCES solicitacoes(id) ON DELETE CASCADE UNIQUE, 
    status VARCHAR(50) DEFAULT 'Em Separação' CHECK (status IN ('Em Separação', 'Concluído', 'Cancelado')),
    data_finalizacao TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Notas Fiscais
CREATE TABLE notas_fiscais (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    solicitacao_id UUID REFERENCES solicitacoes(id) ON DELETE CASCADE, 
    numero_nf VARCHAR(100) NOT NULL,
    data_emissao DATE,
    data_recebimento DATE,
    doc_compras VARCHAR(100),
    valor_total NUMERIC(15, 2),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Tabela de Anexos
CREATE TABLE anexos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    solicitacao_id UUID REFERENCES solicitacoes(id) ON DELETE CASCADE, 
    nome_arquivo VARCHAR(255) NOT NULL,
    url_arquivo TEXT NOT NULL,
    origem VARCHAR(50) DEFAULT 'cliente' CHECK (origem IN ('cliente', 'logistica')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);
